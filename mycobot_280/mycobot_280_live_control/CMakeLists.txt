cmake_minimum_required(VERSION 3.16)
project(mycobot_280_live_control)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(deps rclcpp rclcpp_action moveit_msgs moveit_core rviz_common pluginlib moveit_ros_planning interactive_markers visualization_msgs sensor_msgs geometry_msgs std_msgs std_srvs trajectory_msgs controller_manager_msgs action_msgs tf2_eigen mycobot_280_control_interfaces)
foreach(dep ${deps})
  find_package(${dep} REQUIRED)
endforeach()
find_package(Qt5 REQUIRED COMPONENTS Widgets Test)
include_directories(include)
add_executable(live_follow_node src/live_follow_node.cpp)
ament_target_dependencies(live_follow_node ${deps})
set(CMAKE_AUTOMOC ON)
add_library(live_follow_panel SHARED src/live_follow_panel.cpp include/mycobot_280_live_control/live_follow_panel.hpp)
ament_target_dependencies(live_follow_panel rclcpp rviz_common pluginlib std_msgs std_srvs)
target_link_libraries(live_follow_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugin_description.xml)
install(TARGETS live_follow_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS live_follow_panel LIBRARY DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_live_follow_panel test/test_live_follow_panel.cpp)
  target_link_libraries(test_live_follow_panel live_follow_panel Qt5::Widgets Qt5::Test)
  ament_target_dependencies(test_live_follow_panel rclcpp rviz_common std_msgs std_srvs)
  target_compile_options(test_live_follow_panel PRIVATE -UNDEBUG)
  add_test(NAME live_follow_panel COMMAND test_live_follow_panel)
  set_tests_properties(live_follow_panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(test_control_session test/test_control_session.cpp)
  target_compile_options(test_control_session PRIVATE -UNDEBUG)
  add_test(NAME control_session COMMAND test_control_session)
endif()
ament_package()
